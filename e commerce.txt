
import java.util.ArrayList;
import java.util.Scanner;

class Product {
    int id;
    String name;
    double price;
    int stock;

    Product(int id, String name, double price, int stock) {
        this.id = id;
        this.name = name;
        this.price = price;
        this.stock = stock;
    }

    void display() {
        System.out.println(id + ". " + name
                + " | Price: Rs." + price
                + " | Stock: " + stock);
    }
}

class CartItem {
    Product product;
    int quantity;

    CartItem(Product product, int quantity) {
        this.product = product;
        this.quantity = quantity;
    }

    double getTotal() {
        return product.price * quantity;
    }
}

public class ECommerce {

    static Scanner sc = new Scanner(System.in);

    static ArrayList<Product> products = new ArrayList<>();
    static ArrayList<CartItem> cart = new ArrayList<>();

    static void loadProducts() {
        products.add(new Product(1, "Laptop", 45000, 5));
        products.add(new Product(2, "Headphones", 1500, 10));
        products.add(new Product(3, "Smart Watch", 2500, 8));
        products.add(new Product(4, "Keyboard", 800, 12));
        products.add(new Product(5, "Mouse", 500, 15));
    }

    static void displayProducts() {
        System.out.println("\n===== AVAILABLE PRODUCTS =====");

        for (Product p : products) {
            p.display();
        }
    }

    static Product findProduct(int id) {
        for (Product p : products) {
            if (p.id == id) {
                return p;
            }
        }
        return null;
    }

    static void addToCart() {
        displayProducts();

        System.out.print("\nEnter product ID: ");
        int id = sc.nextInt();

        Product product = findProduct(id);

        if (product == null) {
            System.out.println("Product not found!");
            return;
        }

        System.out.print("Enter quantity: ");
        int quantity = sc.nextInt();

        if (quantity <= 0) {
            System.out.println("Quantity must be greater than zero.");
            return;
        }

        int existingQuantity = 0;

        for (CartItem item : cart) {
            if (item.product.id == id) {
                existingQuantity = item.quantity;
                break;
            }
        }

        if (existingQuantity + quantity > product.stock) {
            System.out.println("Not enough stock available!");
            return;
        }

        boolean found = false;

        for (CartItem item : cart) {
            if (item.product.id == id) {
                item.quantity += quantity;
                found = true;
                break;
            }
        }

        if (!found) {
            cart.add(new CartItem(product, quantity));
        }

        System.out.println("Product added to cart successfully!");
    }

    static double displayCart() {
        System.out.println("\n===== YOUR SHOPPING CART =====");

        if (cart.isEmpty()) {
            System.out.println("Your cart is empty.");
            return 0;
        }

        double total = 0;

        for (CartItem item : cart) {
            System.out.println(
                    item.product.name
                    + " | Quantity: " + item.quantity
                    + " | Price: Rs." + item.getTotal()
            );

            total += item.getTotal();
        }

        System.out.println("------------------------------");
        System.out.println("Total Amount: Rs." + total);

        return total;
    }

    static void checkout() {
        if (cart.isEmpty()) {
            System.out.println("Cart is empty. Add products first.");
            return;
        }

        double total = displayCart();

        System.out.print("\nEnter customer name: ");
        sc.nextLine();
        String customerName = sc.nextLine();

        System.out.println("\n===== ORDER CONFIRMATION =====");
        System.out.println("Customer: " + customerName);
        System.out.println("Total: Rs." + total);

        System.out.print("Confirm order? (yes/no): ");
        String choice = sc.nextLine();

        if (choice.equalsIgnoreCase("yes")) {

            for (CartItem item : cart) {
                item.product.stock -= item.quantity;
            }

            System.out.println("\nOrder placed successfully!");
            System.out.println("Thank you for shopping, " + customerName + "!");

            cart.clear();

        } else {
            System.out.println("Order cancelled.");
        }
    }

    public static void main(String[] args) {

        loadProducts();

        int choice;

        do {
            System.out.println("\n===== E-COMMERCE APPLICATION =====");
            System.out.println("1. View Products");
            System.out.println("2. Add Product to Cart");
            System.out.println("3. View Shopping Cart");
            System.out.println("4. Checkout");
            System.out.println("5. Exit");
            System.out.print("Enter your choice: ");

            if (!sc.hasNextInt()) {
                System.out.println("Please enter a valid number.");
                sc.next();
                continue;
            }

            choice = sc.nextInt();

            switch (choice) {
                case 1:
                    displayProducts();
                    break;

                case 2:
                    addToCart();
                    break;

                case 3:
                    displayCart();
                    break;

                case 4:
                    checkout();
                    break;

                case 5:
                    System.out.println("Thank you! Visit again.");
                    break;

                default:
                    System.out.println("Invalid choice. Try again.");
            }

        } while (choice != 5);

        sc.close();
    }
}
